CREATE DATABASE inasistencias;
use inasistencias;

CREATE TABLE usuarios (
ci varchar(10) not null, 
pass varchar(20) not null,
PRIMARY key (ci)
);

CREATE TABLE inasistencias (
id int(20) AUTO_INCREMENT, 
fechaInicio date not null, 
fechaFin date not null,
materia varchar(20) not null,
grupo varchar(20) not null,
PRIMARY key (id)
);

CREATE TABLE docentes (
ci varchar(10) not null, 
nombre varchar(20) not null, 
apellido varchar(20) not null,
PRIMARY key (ci)
foreign key (ci) references usuarios(ci)
);

CREATE TABLE inasistenciaDocente (
idInasistencia int(20) not null, 
ciDocente varchar(10) not null,
PRIMARY key (idInasistencia, ciDocente),
FOREIGN key (idInasistencia) REFERENCES inasistencias(id),
FOREIGN key (ciDocente) REFERENCES docentes(ci)
);
